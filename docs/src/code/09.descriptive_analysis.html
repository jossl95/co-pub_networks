filter_valid_publication_types = function(data){
  # define list of valid data types
  valid_publication_types <- c(
    'article', 'book', 'book-chapter', 'dissertation', 
    'editorial', 'report', 'review',
    # optional, to consider:
    'erratum', 'letter', 'other', 'preprint'
  )

  # using `purrr` function mapping to select valid types in list of dfs
  data[["works"]] <- data[["works"]] |>
    map(~ {
      # skip elements that are not data frames / tibbles
      if (is.null(.x) || !is.data.frame(.x)) {
        return(.x)
      }
      
      # if there is no `type` column, just return as-is
      if (!"type" %in% names(.x)) {
        return(.x)
      }
      
      filter(.x, type %in% valid_publication_types)
    })

  # filter edge_list
  data[["edge_table"]] <- data[["edge_table"]] |>
    filter(type %in% valid_publication_types)

  return(data)
}
coalesce_scholar_data = function(data){
  # define priority list
  priority <- c("UVA","EUR","RUG","VU","Leiden","RU","UU","Tilburg")

  # coalesce and harmonize position and university
  data[['demographics']] <- data[['demographics']] %>%
    mutate(
      position = coalesce(!!!select(., starts_with("position_"))),
      university = coalesce(!!!select(., starts_with("university_")))
    ) |>
    mutate(
      s = str_to_lower(str_squish(university)),
      university = case_when(
        str_detect(s, "\\brug\\b|groningen") ~ "RUG",
        str_detect(s, "\\bvu\\b|vrij\\s*e?\\s*universiteit") ~ "VU",
        str_detect(s, "\\buva\\b|universiteit van amsterdam|univ(?:ersity)? of amsterdam") ~ "UVA",
        str_detect(s, "\\beur\\b|erasmus") ~ "EUR",
        str_detect(s, "\\bleiden\\b") ~ "LU",
        str_detect(s, "\\bru\\b|radboud") ~ "RU",
        str_detect(s, "\\buu\\b|utrecht") ~ "UU",
        str_detect(s, "\\buvt\\b|universiteit tilburg|\\btilburg\\b") ~ "TU",
        TRUE ~ NA_character_
      )
    )
    
  return(data) 
}
limit_observation_period = function(data, start_date = "2019-01-01"){
  data[['edge_table']] = data[['edge_table']] |>
    filter(publication_date > ymd(start_date))

  return(data)
}
build_matrices <- function(data, 
                           directed_type = NULL, sign_incidence = TRUE, exclude_isolates = TRUE, exclude_edge_with_self=TRUE) {
  stopifnot("uid" %in% colnames(data[['demographics']]))
  stopifnot(all(c("uid", "a__uid") %in% colnames(data[['edge_table']])))

  # 1) create list of node uids
  N_uids <- data[['demographics']] |> distinct(uid) |> pull()

  # 2)clean edges
  E <- data[['edge_table']] |>
    mutate(uid = as.character(uid), a__uid = as.character(a__uid)) |>
    filter(uid %in% N_uids, a__uid %in% N_uids) |>
    relocate(a__uid, .after = uid)

  is_directed <- if(!is.null(directed_type)) TRUE else FALSE

  E = if(exclude_edge_with_self) E |> filter(uid != a__uid) else E
  E = if(is_directed) E |> filter(!(a__author_position == directed_type)) else E

  N_uids <- E |> distinct(uid) |> pull()

  # 3) build graph from edge table
  g <- graph_from_data_frame(
      E,
      directed = !is_directed,
      # vertices = tibble(name = N_uids)
    )

  # (optional) excludes isolates
  if (exclude_isolates) {
      deg <- degree(g, mode = if (is_directed) "all" else "total")
      g   <- induced_subgraph(g, vids = which(deg > 0))
    }

  N_uids <- igraph::V(g)$name
  n      <- length(N_uids)

  # 4) adjacency matrix
  A <- igraph::as_adjacency_matrix(g, sparse = TRUE)
  if (!is_directed) A <- Matrix::forceSymmetric(A, uplo = "U")

  # 5) incidence matrix
  el <- igraph::as_data_frame(g, what = "edges")
  m  <- nrow(el)

  if (m == 0L) {
    B <- sparseMatrix(
      i = integer(), j = integer(), x = numeric(),
      dims     = c(n, 0),
      dimnames = list(N_uids, character())
    )
  } else {
    tail_idx <- match(el$from, N_uids)
    head_idx <- match(el$to,   N_uids)

    vals <- if (is_directed && sign_incidence) {
      c(rep(-1, m), rep(+1, m))
    } else {
      rep(1, 2L * m)
    }

    B <- sparseMatrix(
      i        = c(tail_idx, head_idx),
      j        = rep(seq_len(m), each = 2L),
      x        = vals,
      dims     = c(n, m),
      dimnames = list(N_uids, paste0("e", seq_len(m)))
    )
  }

  # 6) ego × alter counts (still counts of edges in raw Eclean)
  E2 <- E |>
    filter(uid %in% N_uids, a__uid %in% N_uids) |>
    mutate(
      uid    = factor(uid,    levels = N_uids),
      a__uid = factor(a__uid, levels = N_uids)
    )

  E_mat <- if (nrow(E2) == 0L) {
    Matrix::Matrix(
      0,
      nrow = n, ncol = n,
      dimnames = list(N_uids, N_uids),
      sparse   = TRUE
    )
  } else {
    Matrix::Matrix(
      stats::xtabs(~ uid + a__uid, E2),
      sparse = TRUE
    )
  }

  E = E2 |> distinct(uid, a__uid, .keep_all = TRUE)

  # 7) filter demographics to nodes kept
  D = data[['demographics']] |> filter(uid %in% N_uids)

  return(list(
    graph = g,
    A     = A,
    B     = B,
    D     = D,
    E     = E,
    E_mat = E_mat
  ))
}

describe_categorical <- function(data, vars) {
  data |>
    select(all_of(vars)) |>
    # long format: variable + value
    tidyr::pivot_longer(
      cols      = everything(),
      names_to  = "variable",
      values_to = "value"
    ) |>
    group_by(variable) |>
    mutate(
      n_total   = n(),
      n_missing = sum(is.na(value)),
      n_nonmiss = n_total - n_missing
    ) |>
    filter(!is.na(value)) |>
    count(variable, value, n_total, n_missing, n_nonmiss, name = "n") |>
    group_by(variable) |>
    mutate(
      prop        = n / n_nonmiss,
      prop_missing = n_missing / n_total
    ) |>
    ungroup() |>
    arrange(variable, desc(n))
}

compute_ego_statistics <- function(nets) {
  g <- nets$graph
  stopifnot(inherits(g, "igraph"))

  is_dir <- igraph::is_directed(g)

  # 1) Basic degrees
  degree_total <- igraph::degree(
    g,
    mode = if (is_dir) "all" else "total"
  )

  degree_in <- if (is_dir) {
    igraph::degree(g, mode = "in")
  } else {
    rep(NA_real_, igraph::vcount(g))
  }

  degree_out <- if (is_dir) {
    igraph::degree(g, mode = "out")
  } else {
    rep(NA_real_, igraph::vcount(g))
  }

  # 2) Betweenness (normalized)
  betweenness <- igraph::betweenness(
    g,
    directed   = is_dir,
    normalized = TRUE
  )

  # 3) Closeness (normalized)
  closeness <- suppressWarnings(
    igraph::closeness(
      g,
      mode       = if (is_dir) "all" else "all",
      normalized = TRUE
    )
  )

  # 4) Burt’s constraint (structural holes / brokerage)
  constraint <- igraph::constraint(g)

  # 5) Local clustering / transitivity (ego-level)
  clustering <- igraph::transitivity(
    g,
    type = if (is_dir) "local" else "localundirected"
  )
  clustering[is.nan(clustering)] <- NA_real_

  # 6) Eigenvector centrality (prestige / popularity via popular neighbors)
  eig <- igraph::eigen_centrality(
    g,
    directed = is_dir,
    scale    = TRUE
  )$vector

  # 7) k-core number (embeddedness in the dense core)
  kcore <- igraph::coreness(
    g,
    mode = if (is_dir) "all" else "all"
  )

  # 8) Assemble into a tibble
  stats <- tibble::tibble(
    uid          = igraph::V(g)$name,
    degree       = as.numeric(degree_total),
    indegree     = as.numeric(degree_in),
    outdegree    = as.numeric(degree_out),
    betweenness  = as.numeric(betweenness),
    closeness    = as.numeric(closeness),
    constraint   = as.numeric(constraint),
    clustering   = as.numeric(clustering),
    eigenvector  = as.numeric(eig),
    kcore        = as.numeric(kcore)
  )

  # 9) Optionally add demographics stored in nets$D
  if (!is.null(nets$D) && "uid" %in% colnames(nets$D)) {
    stats <- nets$D |>
      right_join(stats, by = "uid") |>
      arrange(uid) |>
      select(-email_22:-discipline_25)
  }

  return(stats)
}

summarise_ego_statistics <- function(ego_stats) {
  # which columns are ego-level stats
  ego_stat_vars <- c(
    "degree",
    "indegree",
    "outdegree",
    "betweenness",
    "closeness",
    "constraint",
    "clustering",
    "eigenvector",
    "kcore"
  )

  # 1) Overall averages for the full sample
  ego_overall <- ego_stats |>
    summarise(
      n = n(),
      across(
        all_of(ego_stat_vars),
        ~ mean(.x, na.rm = TRUE)
      )
    ) |>
    mutate(
      attribute = "scholars",
      level = "all",
      .before = 1
    )

  # 2) Helper for grouped averages (group_var is a *string*)
  summarise_ego_by <- function(data, group_var) {
    res <- data |>
      group_by(.data[[group_var]]) |>
      summarise(
        n = n(),
        across(
          all_of(ego_stat_vars),
          ~ mean(.x, na.rm = TRUE)
        ),
        .groups = "drop"
      )

    res |>
      mutate(
        attribute = group_var,
        level     = .data[[group_var]]
      ) |>
      select(
        attribute,
        level,
        everything(),
        -all_of(group_var)
      )
  }

  # 3) All groupings stacked in one table
  ego_by_all <- bind_rows(
    summarise_ego_by(ego_stats, "gender"),
    summarise_ego_by(ego_stats, "origin"),
    summarise_ego_by(ego_stats, "position"),
    summarise_ego_by(ego_stats, "university")
  )

  # 4) Return as a list
  return(list(
    ego_overall = ego_overall,
    ego_by_all  = ego_by_all
  ))
}

compute_network_statistics <- function(nets) {
  g <- nets$graph
  stopifnot(inherits(g, "igraph"))
  
  is_dir <- igraph::is_directed(g)
  
  # Optionally copy attributes from nets$D onto vertices
  if (!is.null(nets$D) && "uid" %in% colnames(nets$D)) {
    D   <- nets$D
    idx <- match(igraph::V(g)$name, D$uid)
    
    for (v in c("gender", "origin", "university", "position")) {
      if (v %in% colnames(D)) {
        vals <- D[[v]][idx]
        g <- igraph::set_vertex_attr(g, v, value = vals)
      }
    }
  }
  
  # --- Core structural properties (full graph) ---
  comp  <- igraph::components(g)
  gc_id <- which.max(comp$csize)
  gc_v  <- which(comp$membership == gc_id)
  g_gc  <- igraph::induced_subgraph(g, gc_v)
  
  n_nodes <- igraph::vcount(g)
  n_edges <- igraph::ecount(g)
  
  density <- igraph::edge_density(g, loops = FALSE)
  avg_degree <- mean(
    igraph::degree(g, mode = if (is_dir) "all" else "total")
  )
  
  trans_global <- igraph::transitivity(
    g,
    type = if (is_dir) "global" else "globalundirected"
  )
  
  trans_local_vec <- igraph::transitivity(
    g,
    type = if (is_dir) "local" else "localundirected"
  )
  trans_local_mean <- mean(trans_local_vec, na.rm = TRUE)
  
  avg_path_gc <- suppressWarnings(
    igraph::mean_distance(g_gc, directed = is_dir)
  )
  diam_gc <- suppressWarnings(
    igraph::diameter(g_gc, directed = is_dir, unconnected = FALSE)
  )
  
  # --- Simple small-world indicators via ER baseline (full graph) ---
  p <- density
  if (is.finite(p) && !is.na(p) && p > 0) {
    g_rand <- igraph::sample_gnp(
      n       = n_nodes,
      p       = p,
      directed = is_dir,
      loops   = FALSE
    )
    
    trans_rand <- igraph::transitivity(
      g_rand,
      type = if (is_dir) "global" else "globalundirected"
    )
    avg_path_rand <- suppressWarnings(
      igraph::mean_distance(g_rand, directed = is_dir)
    )
    
    smallworld_C_ratio <- trans_global / trans_rand
    smallworld_L_ratio <- avg_path_gc / avg_path_rand
  } else {
    smallworld_C_ratio <- NA_real_
    smallworld_L_ratio <- NA_real_
  }
  
  overall <- tibble::tibble(
    # Size / connectivity
    attribute            = "scholars",
    level                = "all",
    n_nodes              = n_nodes,
    n_edges              = n_edges,
    density              = density,
    avg_degree           = avg_degree,
    
    # Clustering & paths
    transitivity_global       = trans_global,
    transitivity_local_mean   = trans_local_mean,
    avg_path_length_giantcomp = avg_path_gc,
    diameter_giantcomp        = diam_gc,
    
    # Small-world-ish indicators (relative to ER baseline)
    smallworld_C_ratio = smallworld_C_ratio,
    smallworld_L_ratio = smallworld_L_ratio
  )
  
  # --- Helper: compute the same stats on an induced subgraph ---
  compute_structural_stats <- function(g_sub) {
    if (igraph::vcount(g_sub) < 2 || igraph::ecount(g_sub) < 1) {
      return(
        list(
          n_nodes                    = igraph::vcount(g_sub),
          n_edges                    = igraph::ecount(g_sub),
          density                    = NA_real_,
          avg_degree                 = NA_real_,
          transitivity_global        = NA_real_,
          transitivity_local_mean    = NA_real_,
          avg_path_length_giantcomp  = NA_real_,
          diameter_giantcomp         = NA_real_,
          smallworld_C_ratio         = NA_real_,
          smallworld_L_ratio         = NA_real_
        )
      )
    }
    
    comp_sub  <- igraph::components(g_sub)
    gc_id_sub <- which.max(comp_sub$csize)
    gc_v_sub  <- which(comp_sub$membership == gc_id_sub)
    g_gc_sub  <- igraph::induced_subgraph(g_sub, gc_v_sub)
    
    n_nodes_sub <- igraph::vcount(g_sub)
    n_edges_sub <- igraph::ecount(g_sub)
    
    density_sub <- igraph::edge_density(g_sub, loops = FALSE)
    avg_degree_sub <- mean(
      igraph::degree(g_sub, mode = if (is_dir) "all" else "total")
    )
    
    trans_global_sub <- igraph::transitivity(
      g_sub,
      type = if (is_dir) "global" else "globalundirected"
    )
    
    trans_local_vec_sub <- igraph::transitivity(
      g_sub,
      type = if (is_dir) "local" else "localundirected"
    )
    trans_local_mean_sub <- mean(trans_local_vec_sub, na.rm = TRUE)
    
    avg_path_gc_sub <- suppressWarnings(
      igraph::mean_distance(g_gc_sub, directed = is_dir)
    )
    diam_gc_sub <- suppressWarnings(
      igraph::diameter(g_gc_sub, directed = is_dir, unconnected = FALSE)
    )
    
    # Small-world baseline within group
    p_sub <- density_sub
    if (is.finite(p_sub) && !is.na(p_sub) && p_sub > 0) {
      g_rand_sub <- igraph::sample_gnp(
        n       = n_nodes_sub,
        p       = p_sub,
        directed = is_dir,
        loops   = FALSE
      )
      trans_rand_sub <- igraph::transitivity(
        g_rand_sub,
        type = if (is_dir) "global" else "globalundirected"
      )
      avg_path_rand_sub <- suppressWarnings(
        igraph::mean_distance(g_rand_sub, directed = is_dir)
      )
      
      smallworld_C_ratio_sub <- trans_global_sub / trans_rand_sub
      smallworld_L_ratio_sub <- avg_path_gc_sub / avg_path_rand_sub
    } else {
      smallworld_C_ratio_sub <- NA_real_
      smallworld_L_ratio_sub <- NA_real_
    }
    
    list(
      n_nodes                    = n_nodes_sub,
      n_edges                    = n_edges_sub,
      density                    = density_sub,
      avg_degree                 = avg_degree_sub,
      transitivity_global        = trans_global_sub,
      transitivity_local_mean    = trans_local_mean_sub,
      avg_path_length_giantcomp  = avg_path_gc_sub,
      diameter_giantcomp         = diam_gc_sub,
      smallworld_C_ratio         = smallworld_C_ratio_sub,
      smallworld_L_ratio         = smallworld_L_ratio_sub
    )
  }
  
  # --- Helper: apply those stats by levels of a vertex attribute ---
  compute_by_attribute <- function(attr_name) {
    vals <- igraph::vertex_attr(g, attr_name)
    if (is.null(vals)) {
      return(tibble::tibble())
    }
    
    vals_chr <- as.character(vals)
    levels_attr <- sort(unique(vals_chr[!is.na(vals_chr)]))
    if (length(levels_attr) == 0) {
      return(tibble::tibble())
    }
    
    res <- lapply(levels_attr, function(lv) {
      vids <- which(vals_chr == lv)
      if (length(vids) < 2) {
        stats <- compute_structural_stats(
          igraph::induced_subgraph(g, vids)
        )
      } else {
        stats <- compute_structural_stats(
          igraph::induced_subgraph(g, vids)
        )
      }
      
      tibble::tibble(
        attribute                 = attr_name,
        level                     = lv,
        n_nodes                   = stats$n_nodes,
        n_edges                   = stats$n_edges,
        density                   = stats$density,
        avg_degree                = stats$avg_degree,
        transitivity_global       = stats$transitivity_global,
        transitivity_local_mean   = stats$transitivity_local_mean,
        avg_path_length_giantcomp = stats$avg_path_length_giantcomp,
        diameter_giantcomp        = stats$diameter_giantcomp,
        smallworld_C_ratio        = stats$smallworld_C_ratio,
        smallworld_L_ratio        = stats$smallworld_L_ratio
      )
    })
    
    bind_rows(res)
  }
  
  by_group <- bind_rows(
    compute_by_attribute("gender"),
    compute_by_attribute("origin"),
    compute_by_attribute("position"),
    compute_by_attribute("university")
  )
  
  # Return both
  list(
    overall  = overall,
    by_group = by_group
  )
}
